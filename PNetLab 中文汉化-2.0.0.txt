    // =========================================================
    // 创建匹配正则
    //
    // 长词条优先
    // 不使用占位符
    //
    // 例如：
    //
    // "STP Port Roles": "STP Port Roles"
    // "Port": "端口"
    // "Roles": "角色"
    //
    // 会优先匹配：
    //
    // STP Port Roles
    //
    // 然后一次性返回：
    //
    // STP Port Roles
    //
    // 不会再次经过 Port / Roles。
    // =========================================================

    const keys = Object.keys(dict).sort(function (a, b) {
        return b.length - a.length;
    });


    function escapeRegExp(text) {
        return text.replace(/[.*+?^${}()|[\]\\]/g, "\\$&");
    }


    const translateRegex = new RegExp(
        keys.map(escapeRegExp).join("|"),
        "g"
    );


    // =========================================================
    // 翻译文本
    //
    // 注意：
    // 翻译结果不会再次参与匹配。
    // 所以不会出现：
    //
    // __PNETLAB_TRANSLATE_xxx__
    //
    // 也不会出现：
    //
    // STP 端口 角色
    //
    // =========================================================

    function translate(text) {
        if (!text) {
            return text;
        }

        return text.replace(
            translateRegex,
            function (match) {
                return dict[match];
            }
        );
    }


    // =========================================================
    // 处理一个元素
    // =========================================================

    function processElement(element) {
        if (!element || element.nodeType !== 1) {
            return;
        }

        const tag = element.tagName;

        if (
            tag === "SCRIPT" ||
            tag === "STYLE" ||
            tag === "TEXTAREA" ||
            tag === "PRE" ||
            tag === "CODE"
        ) {
            return;
        }


        // -------------------------
        // 直接文本节点
        // -------------------------

        for (let i = 0; i < element.childNodes.length; i++) {
            const node = element.childNodes[i];

            if (node.nodeType === 3) {

                const oldText = node.nodeValue;
                const newText = translate(oldText);

                if (oldText !== newText) {
                    node.nodeValue = newText;
                }
            }
        }


        // -------------------------
        // HTML 属性
        // -------------------------

        const attrs = [
            "title",
            "placeholder",
            "aria-label",
            "alt"
        ];

        attrs.forEach(function (attr) {

            if (!element.hasAttribute(attr)) {
                return;
            }

            const oldValue = element.getAttribute(attr);
            const newValue = translate(oldValue);

            if (oldValue !== newValue) {
                element.setAttribute(
                    attr,
                    newValue
                );
            }

        });

    }


    // =========================================================
    // 初始扫描
    //
    // 页面第一次加载时执行一次
    // =========================================================

    function scan() {

        if (!document.body) {
            return;
        }

        processElement(document.body);

        document.body
            .querySelectorAll("*")
            .forEach(function (element) {
                processElement(element);
            });

    }


    // =========================================================
    // 第一次扫描
    // =========================================================

    scan();


    // =========================================================
    // 监听 PNetLab SPA 动态内容
    //
    // 只处理新增 DOM
    // 不监听 characterData
    // =========================================================

    const observer = new MutationObserver(
        function (mutations) {

            mutations.forEach(function (mutation) {

                mutation.addedNodes.forEach(
                    function (node) {

                        if (node.nodeType === 1) {

                            processElement(node);

                            node.querySelectorAll("*")
                                .forEach(
                                    function (element) {
                                        processElement(element);
                                    }
                                );
                        }

                    }
                );

            });

        }
    );


    observer.observe(
        document.body,
        {
            childList: true,
            subtree: true
        }
    );


    // =========================================================
    // 启动提示
    // =========================================================

    console.log(
        "[PNetLab 汉化] 已启动"
    );

})();
