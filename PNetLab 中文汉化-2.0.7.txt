    // =========================================================
    // 翻译引擎（动态构建）
    // =========================================================
    let entries = [];
    let translateRegex = null;

    function escapeRegExp(text) {
        return text.replace(/[.*+?^${}()|[\]\\]/g, "\\$&");
    }

    function buildSource(key) {
        return escapeRegExp(key)
            .replace(/"/g, '["\\u201C\\u201D]')
            .replace(/ +/g, '\\s+');
    }

    // 根据当前 URL 重新合并字典并构建正则
    function updateTranslator() {
        const url = location.href;
        let mergedDict = {};

        for (let i = 0; i < dictScopes.length; i++) {
            const scope = dictScopes[i];
            if (scope.match.test(url)) {
                Object.assign(mergedDict, scope.dict);
            }
        }

        const keys = Object.keys(mergedDict).sort(function (a, b) {
            return b.length - a.length;
        });

        entries = keys.map(function (key) {
            const source = buildSource(key);
            return {
                source: source,
                testRe: new RegExp("^" + source + "$"),
                value: mergedDict[key]
            };
        });

        translateRegex = entries.length
            ? new RegExp(entries.map(function (e) { return e.source; }).join("|"), "g")
            : null;
    }

    function translate(text) {
        if (!text || !translateRegex) return text;
        return text.replace(translateRegex, function (match) {
            for (let i = 0; i < entries.length; i++) {
                if (entries[i].testRe.test(match)) return entries[i].value;
            }
            return match;
        });
    }

    // =========================================================
    // 跳过区域与匹配规则
    // =========================================================
    const SKIP_TAGS = new Set(["SCRIPT", "STYLE", "TEXTAREA", "PRE", "CODE"]);
    const ATTRS = ["title", "placeholder", "aria-label", "alt"];

    function isSkipped(el) {
        let node = el;
        while (node && node.nodeType === 1) {
            if (SKIP_TAGS.has(node.tagName)) return true;
            node = node.parentNode;
        }
        return false;
    }

    const writtenText = new WeakMap();
    const writtenAttr = new WeakMap();
    const writtenWholeText = new WeakMap();

    const textQueue = new Set();
    const elemQueue = new Set();
    let scheduled = false;

    function queueTextNode(node) {
        if (node && node.nodeType === 3 && node.isConnected) textQueue.add(node);
    }
    function queueElement(el) {
        if (el && el.nodeType === 1 && el.isConnected) elemQueue.add(el);
    }

    function collectTree(root) {
        queueElement(root);
        const walker = document.createTreeWalker(
            root,
            NodeFilter.SHOW_ELEMENT | NodeFilter.SHOW_TEXT,
            null
        );
        let n = walker.nextNode();
        while (n) {
            if (n.nodeType === 1) queueElement(n);
            else if (n.nodeType === 3) queueTextNode(n);
            n = walker.nextNode();
        }
    }

    function processTextNode(node) {
        if (!node.isConnected) return;
        const parent = node.parentNode;
        if (!parent || isSkipped(parent)) return;

        const oldText = node.nodeValue;
        if (writtenText.get(node) === oldText) return;

        const newText = translate(oldText);
        if (oldText !== newText) {
            writtenText.set(node, newText);
            node.nodeValue = newText;
        }
    }

    function processElementAttrs(el) {
        if (!el.isConnected || isSkipped(el)) return;

        let written = writtenAttr.get(el);
        for (let i = 0; i < ATTRS.length; i++) {
            const attr = ATTRS[i];
            if (!el.hasAttribute(attr)) continue;

            const oldValue = el.getAttribute(attr);
            if (written && written[attr] === oldValue) continue;

            const newValue = translate(oldValue);
            if (oldValue !== newValue) {
                if (!written) {
                    written = {};
                    writtenAttr.set(el, written);
                }
                written[attr] = newValue;
                el.setAttribute(attr, newValue);
            }
        }
    }

    const INLINE_TAGS = new Set([
        "SPAN", "B", "STRONG", "I", "EM", "U", "S", "SMALL", "CODE", "A", "MARK",
        "SUP", "SUB", "FONT", "CITE", "Q", "ABBR", "BDI", "BDO", "BR", "WBR",
        "TIME", "VAR", "KBD", "SAMP", "DFN", "DEL", "INS", "LABEL", "STRIKE", "TT", "BIG"
    ]);

    function hasNonInlineChild(el) {
        for (let c = el.firstElementChild; c; c = c.nextElementSibling) {
            if (!INLINE_TAGS.has(c.tagName)) return true;
            if (hasNonInlineChild(c)) return true;
        }
        return false;
    }

    function hasIconOrSvg(el) {
        if (el.tagName === 'SVG' || el.tagName === 'IMG') return true;
        const cls = el.getAttribute('class') || '';
        if (/(^|\s|-)icon|(^|\s)(fa|fas|far|fab|mdi|bi|glyphicon|material-icons)/i.test(cls)) return true;
        for (let c = el.firstElementChild; c; c = c.nextElementSibling) {
            if (hasIconOrSvg(c)) return true;
        }
        return false;
    }

    const INTERACTIVE_SEL = "a,button,input,select,textarea,[role='button'],[onclick]";

    function processElementWholeText(el) {
        if (!el.isConnected || isSkipped(el)) return;
        if (!el.childElementCount) return;
        if (writtenWholeText.get(el) === el.textContent) return;
        if (hasNonInlineChild(el)) return;
        if (el.querySelector(INTERACTIVE_SEL)) return;

        const oldText = el.textContent;
        if (!oldText || oldText.length > 1000) return;

        const trimmed = oldText.trim();
        if (trimmed.length < 15) return;

        if (hasIconOrSvg(el)) return;

        let isFullMatch = false;
        for (let i = 0; i < entries.length; i++) {
            if (entries[i].testRe.test(trimmed)) {
                isFullMatch = true;
                break;
            }
        }
        if (!isFullMatch) return;

        const newText = translate(oldText);
        if (oldText !== newText) {
            writtenWholeText.set(el, newText);
            el.textContent = newText;
            const t = el.firstChild;
            if (t && t.nodeType === 3) writtenText.set(t, newText);
        }
    }

    function flush() {
        scheduled = false;

        if (elemQueue.size) {
            const elems = Array.from(elemQueue);
            elemQueue.clear();
            for (let i = 0; i < elems.length; i++) processElementAttrs(elems[i]);
            for (let i = 0; i < elems.length; i++) processElementWholeText(elems[i]);
        }

        if (textQueue.size) {
            const texts = Array.from(textQueue);
            textQueue.clear();
            for (let i = 0; i < texts.length; i++) processTextNode(texts[i]);
        }
    }

    function schedule() {
        if (scheduled) return;
        scheduled = true;
        if (document.hidden) setTimeout(flush, 0);
        else requestAnimationFrame(flush);
    }

    const observer = new MutationObserver(function (mutations) {
        for (let i = 0; i < mutations.length; i++) {
            const m = mutations[i];
            if (m.type === "characterData") {
                queueTextNode(m.target);
            } else if (m.type === "attributes") {
                queueElement(m.target);
            } else if (m.type === "childList") {
                for (let j = 0; j < m.addedNodes.length; j++) {
                    const node = m.addedNodes[j];
                    if (node.nodeType === 3) queueTextNode(node);
                    else if (node.nodeType === 1) collectTree(node);
                }
            }
        }
        schedule();
    });

    // =========================================================
    // 路由监听与初始化
    // =========================================================
    function onRouteChange() {
        updateTranslator(); // 根据新路由更新字典
        if (document.body) {
            collectTree(document.body); // 重新扫描现有 DOM
            schedule();
        }
    }

    // 拦截 history API，实现 SPA 路由切换感知
    window.addEventListener('popstate', onRouteChange);
    window.addEventListener('hashchange', onRouteChange);

    const originalPushState = history.pushState;
    history.pushState = function () {
        originalPushState.apply(this, arguments);
        setTimeout(onRouteChange, 30); // 延迟等待组件渲染
    };

    const originalReplaceState = history.replaceState;
    history.replaceState = function () {
        originalReplaceState.apply(this, arguments);
        setTimeout(onRouteChange, 30);
    };

    function start() {
        if (!document.body) {
            requestAnimationFrame(start);
            return;
        }

        updateTranslator();
        collectTree(document.body);
        schedule();

        observer.observe(document.documentElement, {
            childList: true,
            subtree: true,
            characterData: true,
            attributes: true,
            attributeFilter: ATTRS
        });

        console.log("[PNetLab 汉化] 已启动 (v2.0.7 动态作用域隔离)");
    }

    start();
